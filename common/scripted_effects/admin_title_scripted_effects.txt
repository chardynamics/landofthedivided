TNO_admin_titles = {
	TNO_admin_title_initialization = yes
	TNO_admin_title_LOTD = yes
	TNO_admin_title_descs = yes

	set_variable = { global.TNO_admin_title_map_mode_config = 1 }
	
	every_country = {
		capital_scope = {
			if = {
				limit = {
					OR = {
						has_variable = AdminTitle
						has_variable = AssocRegion
						has_variable = SubAdminTitle 
					}
				}
				PREV = { set_variable = { TNO_admin_title_map_mode_selected_state = capital } }
			}
			else = {
				set_variable = { PREV.TNO_admin_title_map_mode_selected_state = 1 }
			}
		}
		force_update_map_mode = { mapmode = tno_admin_title_map_mode }
	}
}

## Changes the admin title of the currently scoped state to the admin title specified in `AdminTitle`
# Ex:
# set_temp_variable = { AdminTitle = <admin title token> }
# TNO_adjust_admin_title = yes
TNO_adjust_admin_title = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_admin_title; Old Title: [?AdminTitle.GetTokenKey], New Title: [?admin_title_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { AdminTitle = admin_title_temp }
	custom_effect_tooltip = adjust_admin_title_tt
}

TNO_adjust_assoc_region = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_assoc_region; Old Title: [?AssocRegion.GetTokenKey], New Title: [?assoc_region_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { AssocRegion = assoc_region_temp }
	custom_effect_tooltip = adjust_assoc_region_tt
}

TNO_adjust_assoc_region_2 = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_assoc_region_2; Old Title: [?AssocRegion2.GetTokenKey], New Title: [?assoc_region_2_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { AssocRegion2 = assoc_region_2_temp }
	custom_effect_tooltip = adjust_assoc_region_2_tt
}

TNO_adjust_sub_admin_title = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_sub_admin_title; Old Title: [?SubAdminTitle.GetTokenKey], New Title: [?sub_admin_title_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { SubAdminTitle = sub_admin_title_temp }
	custom_effect_tooltip = adjust_sub_admin_title_tt
}

TNO_clear_admin_title = {
