## Changes the admin title of the currently scoped state to the admin title specified in `AdminTitle`
# Ex:
# set_temp_variable = { AdminTitle = <admin title token> }
# TNO_adjust_admin_title = yes
TNO_adjust_admin_title = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_admin_title; Old Title: [?AdminTitle.GetTokenKey], New Title: [?admin_title_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { AdminTitle = admin_title_temp }
	custom_effect_tooltip = adjust_admin_title_tt
}

TNO_adjust_assoc_region = {
	log = "[GetDateText]: [Root.GetName]: TNO_adjust_assoc_region; Old Title: [?AssocRegion.GetTokenKey], New Title: [?assoc_region_temp.GetTokenKey], State [THIS.GetID]"
	set_variable = { AssocRegion = assoc_region_temp }
	custom_effect_tooltip = adjust_assoc_region_tt
}

TNO_admin_title_state_control_effect = {
	if = {
		limit = {
			FROM.FROM = {
				OR = {
					state = 706
					state = 707 
				}
			}
		}
		FROM.FROM = {
			set_temp_variable = { admin_title_temp = token:Admin_Title_Military_Territory }
			TNO_adjust_admin_title = yes
		}
	}
	else_if = { #Failsafe for WW3
		limit = {
			FROM = {
				tag = JAP
			}
			ROOT = {
				tag = USA 
			}
			JAP = {
				has_war_with = USA
			}
			USA = {
				has_war_with = JAP
			}
		}
		if = {
			limit = {
				FROM.FROM = {
					state = 629
					NOT = { is_controlled_by = USA }
				}
			}
			FROM.FROM = {
				clear_variable = TNO_assoc_region 
				set_temp_variable = { admin_title_temp = token:Admin_Title_State }
				TNO_adjust_admin_title = yes
				set_temp_variable = { sub_admin_title_temp = token:Admin_Title_County }
				TNO_adjust_sub_admin_title = yes
			}
		}
		if = {
			limit = {
				FROM.FROM = {
					OR = {
						state = 768
						state = 769
					}
					NOT = { is_controlled_by = USA }
				}
			}
			FROM.FROM = {
				set_temp_variable = { admin_title_temp = token:Admin_Title_Port }
				TNO_adjust_admin_title = yes
			}
		}
		if = {
			limit = {
				FROM.FROM = {
					OR = {
						state = 2518
						state = 2523
					}
					NOT = { is_controlled_by = USA }
				}
			}
			clear_variable = TNO_assoc_region 
			FROM.FROM = {
				set_temp_variable = { admin_title_temp = token:Admin_Title_Unc_Territory }
				TNO_adjust_admin_title = yes
			}
		}
	}
}
