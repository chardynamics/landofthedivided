IBR_rerender_democratic_council_support = {
	clear_array = demIberianCouncil_seats

	#Part that supports bill (moves 20 pixels to the left)
	set_temp_variable = { total = 0 }
	for_loop_effect = {
		start = 0
		end = 11
		value = i

		set_temp_variable = { ir = i }

		set_temp_variable = { old_total = total }
		log = "i: [?i]"
		log = "ir: [?ir]"
		log = "old_total: [?old_total]"
		add_to_temp_variable = { total = IBR_democratic_council_faction_seats_supporting^ir }
		log = "total: [?total]"
		log = "supporting: [?IBR_democratic_council_faction_seats_supporting^ir]"
		resize_array = {
			array = demIberianCouncil_seats
			value = ir
			size = total
		}

		for_loop_effect = {
			start = old_total
			end = total
			value = k

			add_to_variable = { demIberianCouncil_seats_x^k = -20 }
		}
	}
	#Part that votes blank on the bill (stays in place)
	for_loop_effect = {
		start = 0
		end = 11
		value = i

		set_temp_variable = { ir = i }

		set_temp_variable = { old_total = total }
		log = "i: [?i]"
		log = "ir: [?ir]"
		log = "old_total: [?old_total]"
		add_to_temp_variable = { total = IBR_democratic_council_faction_seats_abstain^ir }
		log = "total: [?total]"
		log = "abstaining: [?IBR_democratic_council_faction_seats_abstain^ir]"
		resize_array = {
			array = demIberianCouncil_seats
			value = ir
			size = total
		}

		for_loop_effect = {
			start = old_total
			end = total
			value = k

			add_to_variable = { demIberianCouncil_seats_x^k = 0 }
		}
	}
	#Part that does not support bill (moves 20 pixels to the right)
	for_loop_effect = {
		start = 0
		end = 11
		value = i

		set_temp_variable = { ir = i }

		set_temp_variable = { old_total = total }
		log = "i: [?i]"
		log = "ir: [?ir]"
		log = "old_total: [?old_total]"

		set_temp_variable = { non_supporting_seats = IBR_democratic_council_faction_seats^ir }
		subtract_from_temp_variable = { non_supporting_seats = IBR_democratic_council_faction_seats_supporting^ir }
		subtract_from_temp_variable = { non_supporting_seats = IBR_democratic_council_faction_seats_abstain^ir }
		add_to_temp_variable = { total = non_supporting_seats }
		log = "total: [?total]"
		log = "not supporting: [?non_supporting_seats]"


		resize_array = {
			array = demIberianCouncil_seats
			value = ir
			size = total
		}

		for_loop_effect = {
			start = old_total
			end = total
			value = k

			add_to_variable = { demIberianCouncil_seats_x^k = 20 }

		}
	}
}

IBR_democratic_council_update_seat_order = {
	#0 Reformists
	#1 Conservatives
	#2 Hardliners
	find_highest_in_array = {
		array = IBR_democratic_council_total_faction_seats
		index = i
	}
	if = { # find_highest_in_array defaults to the first index in case of a tie, so 2 must be larger than 0 and 1, but 1 may be equal to 2, and 0 could be tied 3 ways
		limit = { check_variable = { i = 2 } }
		if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^0 > IBR_democratic_council_total_faction_seats^1 } }
			set_variable = { IBR_democratic_council_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 0 }
			set_variable = { IBR_democratic_council_last_group = 1 }
		}
		else = {
			set_variable = { IBR_democratic_council_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 1 }
			set_variable = { IBR_democratic_council_last_group = 0 }
		}
	}
	else_if = {
		limit = { check_variable = { i = 1 } }
		if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^1 = IBR_democratic_council_total_faction_seats^2 } }
			set_variable = { IBR_democratic_council_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 1 }
			set_variable = { IBR_democratic_council_last_group = 0 }
		}
		else_if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^0 > IBR_democratic_council_total_faction_seats^2 } }
			set_variable = { IBR_democratic_council_biggest_group = 1 }
			set_variable = { IBR_democratic_council_second_biggest_group = 0 }
			set_variable = { IBR_democratic_council_last_group = 2 }
		}
		else = {
			set_variable = { IBR_democratic_council_biggest_group = 1 }
			set_variable = { IBR_democratic_council_second_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 0 }
		}
	}
	else = {
		if = {
			limit = {
				check_variable = { IBR_democratic_council_total_faction_seats^0 = IBR_democratic_council_total_faction_seats^1 }
				check_variable = { IBR_democratic_council_total_faction_seats^0 = IBR_democratic_council_total_faction_seats^2 }
			}
			set_variable = { IBR_democratic_council_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 1 }
			set_variable = { IBR_democratic_council_last_group = 0 }
		}
		else_if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^0 = IBR_democratic_council_total_faction_seats^2 } }
			set_variable = { IBR_democratic_council_biggest_group = 2 }
			set_variable = { IBR_democratic_council_second_biggest_group = 0 }
			set_variable = { IBR_democratic_council_last_group = 1 }
		}
		else_if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^0 = IBR_democratic_council_total_faction_seats^1 } }
			set_variable = { IBR_democratic_council_biggest_group = 1 }
			set_variable = { IBR_democratic_council_second_biggest_group = 0 }
			set_variable = { IBR_democratic_council_last_group = 2 }
		}
		else_if = {
			limit = { check_variable = { IBR_democratic_council_total_faction_seats^1 > IBR_democratic_council_total_faction_seats^2 } }
			set_variable = { IBR_democratic_council_biggest_group = 0 }
			set_variable = { IBR_democratic_council_second_biggest_group = 1 }
			set_variable = { IBR_democratic_council_last_group = 2 }
		}
		else = {
			set_variable = { IBR_democratic_council_biggest_group = 0 }
			set_variable = { IBR_democratic_council_second_biggest_group = 2 }
			set_variable = { IBR_democratic_council_last_group = 1 }
		}
	}
}

IBR_democratic_council_distribute_subfaction_seats = {
	if = {
		limit = {
			NOT = {
				has_variable = IBR_democratic_council_independent_proportion
			}
		}
		set_variable = { IBR_democratic_council_independent_proportion = 0.1 }
	}
	else = {
		set_variable = { IBR_democratic_council_independent_proportion = IBR_democratic_council_faction_seats^IBR_democratic_council_PCE_LL_subfaction }
		divide_variable = { IBR_democratic_council_independent_proportion = IBR_democratic_council_total_faction_seats^0 }
	}
	if = {
		limit = {
			NOT = {
				has_variable = IBR_democratic_council_catholic_reformer_proportion
			}
		}
		set_variable = { IBR_democratic_council_catholic_reformer_proportion = 0.25 }
	}
	else = {
		set_variable = { IBR_democratic_council_catholic_reformer_proportion = IBR_democratic_council_faction_seats^IBR_democratic_council_PCP_FEPU_subfaction }
		divide_variable = { IBR_democratic_council_catholic_reformer_proportion = IBR_democratic_council_total_faction_seats^0 }
	}
	if = {
		limit = {
			NOT = {
				has_variable = IBR_democratic_council_blue_reformer_proportion
			}
		}
		set_variable = { IBR_democratic_council_blue_reformer_proportion = 0.3 }
	}
	else = {
		set_variable = { IBR_democratic_council_blue_reformer_proportion = IBR_democratic_council_faction_seats^IBR_democratic_council_FPS_subfaction }
		divide_variable = { IBR_democratic_council_blue_reformer_proportion = IBR_democratic_council_total_faction_seats^0 }
	}

	set_temp_variable = { IBR_democratic_council_reformist_seats_assigned = 0 }

	set_temp_variable = { IBR_democratic_council_independent_seats = IBR_democratic_council_total_faction_seats^0 }
	multiply_temp_variable = { IBR_democratic_council_independent_seats = IBR_democratic_council_independent_proportion }
	round_temp_variable = IBR_democratic_council_independent_seats
	add_to_variable = { IBR_democratic_council_faction_seats^IBR_democratic_council_PCE_LL_subfaction = IBR_democratic_council_independent_seats }
	add_to_temp_variable = { IBR_democratic_council_reformist_seats_assigned = IBR_democratic_council_independent_seats }

	set_temp_variable = { IBR_democratic_council_catholic_reformer_seats = IBR_democratic_council_total_faction_seats^0 }
	multiply_temp_variable = { IBR_democratic_council_catholic_reformer_seats = IBR_democratic_council_catholic_reformer_proportion }
	round_temp_variable = IBR_democratic_council_catholic_reformer_seats
	add_to_variable = { IBR_democratic_council_faction_seats^IBR_democratic_council_PCP_FEPU_subfaction = IBR_democratic_council_catholic_reformer_seats }
	add_to_temp_variable = { IBR_democratic_council_reformist_seats_assigned = IBR_democratic_council_catholic_reformer_seats }

	set_temp_variable = { IBR_democratic_council_blue_reformer_seats = IBR_democratic_council_total_faction_seats^0 }
	multiply_temp_variable = { IBR_democratic_council_blue_reformer_seats = IBR_democratic_council_blue_reformer_proportion }
	round_temp_variable = IBR_democratic_council_blue_reformer_seats
	add_to_variable = { IBR_democratic_council_faction_seats^IBR_democratic_council_FPS_subfaction = IBR_democratic_council_blue_reformer_seats }
	add_to_temp_variable = { IBR_democratic_council_reformist_seats_assigned = IBR_democratic_council_blue_reformer_seats }

	set_temp_variable = { IBR_democratic_council_modernist_seats = IBR_democratic_council_total_faction_seats^0 }
	subtract_from_temp_variable = { IBR_democratic_council_modernist_seats = IBR_democratic_council_reformist_seats_assigned }
	add_to_variable = { IBR_democratic_council_faction_seats^IBR_democratic_council_EC_subfaction = IBR_democratic_council_modernist_seats }

	if = {
		limit = {
			NOT = {
				has_variable = IBR_democratic_council_moderate_conservative_proportion
			}
		}
		set_variable = { IBR_democratic_council_moderate_conservative_proportion = 0.3 }
	}
	else = {
		set_variable = { IBR_democratic_council_moderate_conservative_proportion = IBR_democratic_council_faction_seats^IBR_democratic_council_EEH_subfaction }
		divide_variable = { IBR_democratic_council_moderate_conservative_proportion = IBR_democratic_council_total_faction_seats^1 }
	}
	if = {
		limit = {
			NOT = {
				has_variable = IBR_democratic_council_technocrat_proportion
			}
		}
		set_variable = { IBR_democratic_council_technocrat_proportion = 0.25 }
	}
	else = {
		set_variable = { IBR_democratic_council_technocrat_proportion = IBR_democratic_council_faction_seats^IBR_democratic_council_EDC_subfaction }
		divide_variable = { IBR_democratic_council_technocrat_proportion = IBR_democratic_council_total_faction_seats^1 }
	}
	if = {

IBR_democratic_council_change_seats = {
	if = {
		limit = {
			check_variable = {
				var = IBR_democratic_council_faction_seats^losing_group
				value = seat_change
				compare = greater_than_or_equals
			}
		}
		subtract_from_variable = { IBR_democratic_council_faction_seats^losing_group = seat_change }
		add_to_variable = { IBR_democratic_council_faction_seats^winning_group = seat_change }
		if = {
			limit = {
				OR = {
					check_variable = { losing_group = 0 }
					check_variable = { losing_group = 1 }
					check_variable = { losing_group = 2 }
					check_variable = { losing_group = 3 }
				}
			}
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^0 = seat_change }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { losing_group = 4 }
					check_variable = { losing_group = 5 }
					check_variable = { losing_group = 6 }
					check_variable = { losing_group = 7 }
				}
			}
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^1 = seat_change }
		}
		else = {
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^2 = seat_change }
		}
		if = {
			limit = {
				OR = {
					check_variable = { winning_group = 0 }
					check_variable = { winning_group = 1 }
					check_variable = { winning_group = 2 }
					check_variable = { winning_group = 3 }
				}
			}
			add_to_variable = { IBR_democratic_council_total_faction_seats^0 = seat_change }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { winning_group = 4 }
					check_variable = { winning_group = 5 }
					check_variable = { winning_group = 6 }
					check_variable = { winning_group = 7 }
				}
			}
			add_to_variable = { IBR_democratic_council_total_faction_seats^1 = seat_change }
		}
		else = {
			add_to_variable = { IBR_democratic_council_total_faction_seats^2 = seat_change }
		}
	}
	else = {
		set_temp_variable = { seat_change = IBR_democratic_council_faction_seats^losing_group }
		subtract_from_variable = { IBR_democratic_council_faction_seats^losing_group = seat_change }
		add_to_variable = { IBR_democratic_council_faction_seats^winning_group = seat_change }
		if = {
			limit = {
				OR = {
					check_variable = { losing_group = 0 }
					check_variable = { losing_group = 1 }
					check_variable = { losing_group = 2 }
					check_variable = { losing_group = 3 }
				}
			}
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^0 = seat_change }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { losing_group = 4 }
					check_variable = { losing_group = 5 }
					check_variable = { losing_group = 6 }
					check_variable = { losing_group = 7 }
				}
			}
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^1 = seat_change }
		}
		else = {
			subtract_from_variable = { IBR_democratic_council_total_faction_seats^2 = seat_change }
		}
		if = {
			limit = {
				OR = {
					check_variable = { winning_group = 0 }
					check_variable = { winning_group = 1 }
					check_variable = { winning_group = 2 }
					check_variable = { winning_group = 3 }
				}
			}
			add_to_variable = { IBR_democratic_council_total_faction_seats^0 = seat_change }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { winning_group = 4 }
					check_variable = { winning_group = 5 }
					check_variable = { winning_group = 6 }
					check_variable = { winning_group = 7 }
				}
			}
			add_to_variable = { IBR_democratic_council_total_faction_seats^1 = seat_change }
		}
		else = {
			add_to_variable = { IBR_democratic_council_total_faction_seats^2 = seat_change }
		}
	}
	if = {
		limit = {
			check_variable = { seat_change > 0 }
		}
