# DESCRIPTION: Initializes the entire mechanic, making it both accessible and active.
# INPUT: n/a
SalvadoranCoup_Initialize = {
	log = "[GetDateText]: [Root.GetTag]: SalvadoranCoup_Initialize; Executing."

	# 0-1, Empty-Full. Float, %.
	set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness = 0.19 }
	set_variable = { ELS.SalvadoranCoup_CivilianAnger = 0.61 }

	# GUI icon frames for the above.
	set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 2 }
	set_variable = { ELS.SalvadoranCoup_CivilianAnger_frame = 4 }

	# Both values below are accamulated, separate integer values.
	if = {
		limit = {
			has_game_rule = {
				rule = ELS_COUP_PATH
				option = ELS_COUP_PATH_YES
			}
		}
		set_variable = { ELS.SalvadoranCoup_PopularityCastrists = 100 }
		set_variable = { ELS.SalvadoranCoup_PopularityHispanists = 0 }
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = ELS_COUP_PATH
				option = ELS_COUP_PATH_PROAXIS
			}
		}
		set_variable = { ELS.SalvadoranCoup_PopularityCastrists = 0 }
		set_variable = { ELS.SalvadoranCoup_PopularityHispanists = 100 }
	}
	else_if = {
		limit = { has_global_flag = PPI_Rebel_Victory }
		set_variable = { ELS.SalvadoranCoup_PopularityCastrists = 60 }
		set_variable = { ELS.SalvadoranCoup_PopularityHispanists = 40 }
	}
	else = {
		set_variable = { ELS.SalvadoranCoup_PopularityCastrists = 40 }
		set_variable = { ELS.SalvadoranCoup_PopularityHispanists = 60 }
	}
	SalvadoranCoup_ComputePopularityFrame = yes

	SalvadoranCoup_SetDecisionVariablesIBR = yes
	SalvadoranCoup_SetDecisionVariablesMEX = yes

	ELS = {
		hidden_effect = {
			set_political_power = 150
			activate_mission = SalvadoranCoup_Deadline
			set_variable = { SalvadoranCoup_PulseDuration = 7 }
			activate_mission = SalvadoranCoup_Pulse
			activate_mission = SalvadoranCoup_Readjustment
		}
	}
	
	USA = { set_country_flag = USA_Operation_Granite_Available }
	
	set_variable = { USA_CIA_delay_salvadoran_coup_count = 0 }
	set_variable = { JAP.ELS_Japan_Activity = -1 }
	
	log = "[GetDateText]: [Root.GetTag]: SalvadoranCoup_Initialize; Completed execution."
}

# DESCRIPTION: Changes the value of Military Preparedness by a custom amount.
# INPUT: change_v
SalvadoranCoup_ChangeMilitaryPreparedness = {
	custom_effect_tooltip = SalvadoranCoup_ChangeMilitaryPreparedness_tt
	add_to_variable = { ELS.SalvadoranCoup_MilitaryPreparedness = change_v }
	if = {
		limit = {
			OR = {
				has_game_rule = {
					rule = ELS_COUP_PATH
					option = ELS_COUP_PATH_NO
				}
				AND = {
					ELS = { has_country_flag = ELS_Salvadoran_Coup_USA_win }
					NOT = {
						has_game_rule = {
							rule = ELS_COUP_PATH
							option = ELS_COUP_PATH_YES
						}
						has_game_rule = {
							rule = ELS_COUP_PATH
							option = ELS_COUP_PATH_PROAXIS
						}
					}
				}
			}
		}
		clamp_variable = { var = ELS.SalvadoranCoup_MilitaryPreparedness min = 0 max = 0.89 }
	}
	else_if = {
		limit = {
			OR = {
				has_game_rule = {
					rule = ELS_COUP_PATH
					option = ELS_COUP_PATH_YES
				}
				has_game_rule = {
					rule = ELS_COUP_PATH
					option = ELS_COUP_PATH_PROAXIS
				}
			}
		}
		clamp_variable = { var = ELS.SalvadoranCoup_MilitaryPreparedness min = 0.9 max = 1 }
	}
	else = {
		clamp_variable = { var = ELS.SalvadoranCoup_MilitaryPreparedness min = 0 max = 1 }
	}

	# Calculating GUI icon frame.
	if = {
		limit = {
			check_variable = { ELS.SalvadoranCoup_MilitaryPreparedness < 0.30 }
		}
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 1 }
	}
	else_if = {
		limit = { 
			check_variable = { ELS.SalvadoranCoup_MilitaryPreparedness < 0.45 }
		}
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 2 }
	}
	else_if = {
		limit = { 
			check_variable = { ELS.SalvadoranCoup_MilitaryPreparedness < 0.60 }
		}
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 3 }
	}
	else_if = {
		limit = { 
			check_variable = { ELS.SalvadoranCoup_MilitaryPreparedness < 0.75 }
		}
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 4 }
	}
	else_if = {
		limit = { 
			check_variable = { ELS.SalvadoranCoup_MilitaryPreparedness < 0.90 } 
		}
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 5 }
	}
	else = {
		set_variable = { ELS.SalvadoranCoup_MilitaryPreparedness_frame = 6 }
	}
}

# DESCRIPTION: Computes the GUI frame of the factionalism pie chart.
# INPUT: n/a
SalvadoranCoup_ComputePopularityFrame = {
	set_temp_variable = { popularity_sum = 0 }
	add_to_temp_variable = { popularity_sum = ELS.SalvadoranCoup_PopularityCastrists }
	add_to_temp_variable = { popularity_sum = ELS.SalvadoranCoup_PopularityHispanists }
	set_variable = { ELS.SalvadoranCoup_Factionalism_frame = ELS.SalvadoranCoup_PopularityCastrists }
	divide_variable = { ELS.SalvadoranCoup_Factionalism_frame = popularity_sum }
	multiply_variable = { ELS.SalvadoranCoup_Factionalism_frame = 100 }
}

# DESCRIPTION: Changes the Military Preparednes based on the value of Salvadoran Stability.
# Part of SalvadoranCoup_Deadline mission timeout effect.
# INPUT: n/a
SalvadoranCoup_ComputePulseMilitaryPreparednessDecrease = {
	# change_v = -(X * Y * Z),
	# where X = ELS Stability / 10, Y = internal factor, Z = external factor
	set_temp_variable = { X = ELS.stability }
	divide_temp_variable = { X = 10 }
	
	set_temp_variable = { Y = 1 }
	if = {
		limit = { check_variable = { ELS.SalvadoranCoup_CivilianAnger > ELS.SalvadoranCoup_MilitaryPreparedness } }
		add_to_temp_variable = { Y = 0.15 }
	}
	if = {
		limit = { ELS = { has_decision = ELS_SalvadoranCoup_CombatTheCombatants } }
		add_to_temp_variable = { Y = 0.3 }
	}

	set_temp_variable = { Z = 1 }
	if = {
		limit = { IBR = { has_decision = IBR_SalvadoranCoup_GiftsAndFavors } }
		if = {
			limit = { check_variable = { IBR.SalvadoranCoup_GiftsAndFavors_CostM < 0.15 } }
			add_to_temp_variable = { Z = -0.15 }
		}
		else = {
			add_to_temp_variable = { Z = -0.30 }
		}
	}

	set_temp_variable = { change_v = X }
	multiply_temp_variable = { change_v = Y }
	multiply_temp_variable = { change_v = Z }
	multiply_temp_variable = { change_v = -1 }
	SalvadoranCoup_ChangeMilitaryPreparedness = yes
}

# DESCRIPTION: Changes the Civilian Anger based on the value of Salvadoran War Support.
# Part of SalvadoranCoup_Deadline mission timeout effect.
# INPUT: n/a
SalvadoranCoup_ComputePulseCivilianAngerDecrease = {
	# change_v = -(X * Y * Z),
	# where X = ELS War Support / 7, Y = internal factor, Z = external factor
	set_temp_variable = { X = ELS.has_war_support }
	divide_temp_variable = { X = 7 }
	
	set_temp_variable = { Y = 1 }
	if = {
		limit = { check_variable = { ELS.days_mission_timeout@SalvadoranCoup_Deadline > 100 } }
		add_to_temp_variable = { Y = -0.2 }
	}
	if = {
		limit = { ELS = { has_decision = ELS_SalvadoranCoup_CombatTheCombatants } }
		add_to_temp_variable = { Y = 0.2 }
	}
	
	set_temp_variable = { Z = 1 }
	if = {
		limit = { MEX = { has_decision = MEX_SalvadoranCoup_AJustCause } }
		add_to_temp_variable = { Z = -0.3 }
	}
	set_temp_variable = { change_v = X }
	
	multiply_temp_variable = { change_v = Y }
	multiply_temp_variable = { change_v = Z }
	multiply_temp_variable = { change_v = -1 }
	SalvadoranCoup_ChangeCivilianAnger = yes
}
