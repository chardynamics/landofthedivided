IBR_wars_calculate_homefront_commodities = {
	set_variable = { hc_production = hc_production_base }
	multiply_variable = { hc_production = modifier@IBR_wars_hc_production_base }
	set_temp_variable = { hc_production_civs = num_of_civilian_factories }
	multiply_temp_variable = { hc_production_civs = 2 }
	add_to_variable = { hc_production = hc_production_civs }
	set_temp_variable = { hc_production_core = num_core_states }
	multiply_temp_variable = { hc_production_core = modifier@IBR_wars_hc_production_core_states }
	add_to_variable = { hc_production = hc_production_core }
	set_temp_variable = { hc_production_occupied = num_occupied_states }
	multiply_temp_variable = { hc_production_occupied = modifier@IBR_wars_hc_production_occupied_states }
	add_to_variable = { hc_production = hc_production_occupied }

	set_variable = { hc_consumption = hc_consumption_base }
	multiply_variable = { hc_consumption = modifier@IBR_wars_hc_consumption_base }
	set_temp_variable = { hc_consumption_pop = max_manpower_k }
	divide_temp_variable = { hc_consumption_pop = 100 }
	multiply_temp_variable = { hc_consumption_pop = modifier@IBR_wars_hc_consumption_population }
	set_temp_variable = { hc_consumption_core = num_core_states }
	multiply_temp_variable = { hc_consumption_core = modifier@IBR_wars_hc_consumption_core_states }
	set_temp_variable = { hc_consumption_occupied = num_occupied_states }
	multiply_temp_variable = { hc_consumption_occupied = modifier@IBR_wars_hc_consumption_occupied_states }
	set_temp_variable = { hc_consumption_states = hc_consumption_core }
	add_to_temp_variable = { hc_consumption_states = hc_consumption_occupied }
	divide_temp_variable = { hc_consumption_pop = hc_consumption_states }
	add_to_variable = { hc_consumption = hc_consumption_pop }

	set_variable = { hc_difference = hc_production }
	subtract_from_variable = { hc_difference = hc_consumption }
	round_variable = hc_difference
	add_to_variable = { homefront_commodities_total = hc_difference }
	round_variable = homefront_commodities_total
	IBR_wars_update_hc_modifiers = yes
}

IBR_Wars_calculate_score = {
	set_temp_variable = { global.IBRWars_USA_points = 0 }
	set_temp_variable = { global.IBRWars_GER_points = 0 }
	every_country = { # Spanish Faction
		limit = { is_spanish_faction = yes }
		if = {
			limit = {
				OR = {
					has_non_aggression_pact_with = USA
					check_variable = { ECON_SPHERE_LEADER = USA }
					check_variable = { ECON_SPHERE_LEADER = BRA }
				}
			}
			add_to_temp_variable = { global.IBRWars_USA_points = 150 }
		}
		if = {
			limit = {
				OR = {
					has_non_aggression_pact_with = GER
					check_variable = { ECON_SPHERE_LEADER = GER }
				}
			}
			add_to_temp_variable = { global.IBRWars_GER_points = 150 }
		}
	}
	every_country = { # Portuguese Faction
		limit = { is_portuguese_faction = yes }
		if = {
			limit = {
				OR = {
					has_non_aggression_pact_with = USA
					check_variable = { ECON_SPHERE_LEADER = USA }
					check_variable = { ECON_SPHERE_LEADER = BRA }
				}
			}
			add_to_temp_variable = { global.IBRWars_USA_points = 75 }
			if = {
				limit = {
					has_subject = GNB
				}
				add_to_temp_variable = { global.IBRWars_USA_points = 25 }
			}
			if = {
				limit = {
					has_subject = CPV
				}
				add_to_temp_variable = { global.IBRWars_USA_points = 15 }
			}
		}
		if = {
			limit = {
				OR = {
					has_non_aggression_pact_with = GER
					check_variable = { ECON_SPHERE_LEADER = GER }
				}
			}
			add_to_temp_variable = { global.IBRWars_GER_points = 75 }
			if = {
				limit = {
					has_subject = GNB
				}
				add_to_temp_variable = { global.IBRWars_GER_points = 25 }
			}
			if = {
				limit = {
					has_subject = CPV
				}
				add_to_temp_variable = { global.IBRWars_GER_points = 15 }
			}
		}
	}

	# Final Score Assignment
	if = {
		limit = {
			check_variable = { global.IBRWars_USA_points > global.IBRWars_GER_points }
		}
		USA = {
			set_temp_variable = { Victory_ID = 18 }
			set_temp_variable = { Victory_Outcome = 4 }
			set_temp_variable = { Victory_Points = global.IBRWars_USA_points }
			Cold_War_GUI_ManipulateVictory = yes
		}
		GER = {
			set_temp_variable = { Victory_ID = 18 }
			set_temp_variable = { Victory_Outcome = 2 }
			set_temp_variable = { Victory_Points = global.IBRWars_GER_points }
			Cold_War_GUI_ManipulateVictory = yes
		}
	}
	else_if = {
		limit = {
			check_variable = { global.IBRWars_USA_points < global.IBRWars_GER_points }
		}
		USA = {
			set_temp_variable = { Victory_ID = 18 }
			set_temp_variable = { Victory_Outcome = 2 }
			set_temp_variable = { Victory_Points = global.IBRWars_USA_points }
