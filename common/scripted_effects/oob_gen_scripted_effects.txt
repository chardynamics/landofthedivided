d_create_army = {
	##Most countries
	division_template = {
		name = "Infantry Unit"

		regiments = {
			infantry = { x = 0 y = 0 }
			infantry = { x = 0 y = 1 }
			infantry = { x = 0 y = 2 }
			infantry = { x = 1 y = 0 }
			infantry = { x = 1 y = 1 }
			infantry = { x = 1 y = 2 }
		}
		support = {
			artillery = { x = 0 y = 0 }
			engineer = { x = 0 y = 1 }
		}
	}

	set_variable = { oob_gen_script_id = 1 }
	oob_gen_reset_spawn_counters = yes

	set_temp_variable = { num_of_divs = max_available_manpower_k }
	divide_temp_variable = { num_of_divs = 10 }
	set_variable = { oob_gen_planned_divs = num_of_divs }
	set_variable = { oob_gen_barrier = 0 }
	set_variable = { oob_gen_j_cutoff = 0 }

	set_temp_variable = { i = 0 }
	while_loop_effect = {
		limit = { check_variable = { i < num_of_divs } }
		random_controlled_state = {
			oob_gen_count_infantry_unit_spawn = yes
			if = {
				limit = { PREV = { has_country_flag = oob_gen_verbose_log } }
				log = "PYTHON_OOB_SPAWN;[PREV.GetTag];1;[?i];Infantry Unit;state=[THIS.GetName];exp=0.5;equip=0.75"
			}
			create_unit = {
				division = "division_template = \"Infantry Unit\" start_experience_factor = 0.5 start_equipment_factor = 0.75"
				owner = PREV
			}
		}
		add_to_temp_variable = { i = 1 }
		multiply_temp_variable = { i = 1 }
	}

	oob_gen_log_summary = yes
}

		multiply_temp_variable = { i = 1.02 }

		}

			capital_scope = {
				oob_gen_count_motorized_division_spawn = yes
				if = {
					limit = { PREV = { has_country_flag = oob_gen_verbose_log } }

	oob_gen_log_summary = yes
}

# Enable per-division PYTHON_OOB_SPAWN lines for one country (console: execute oob_gen_enable_verbose_log)
oob_gen_enable_verbose_log = {
	set_country_flag = oob_gen_verbose_log
	log = "PYTHON_OOB_VERBOSE_ENABLED;[GetTag];[GetName]"
}

oob_gen_disable_verbose_log = {
	clr_country_flag = oob_gen_verbose_log
	log = "PYTHON_OOB_VERBOSE_DISABLED;[GetTag];[GetName]"
}
