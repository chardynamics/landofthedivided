TNO_fopo_setup_caracas_nights = {
	set_global_flag = VEN_CN_ongoing
	hidden_effect = {
		ZZZ = {
			activate_mission = ZZZ_CN_dummy
		}
	}
	activate_mission = IBR_CN_coup_countdown
	activate_mission = IBR_CN_time_mission
	set_variable = { global.ven_coup_military_readiness = 35 }
	set_variable = { global.ven_coup_popularity = 15 }
	set_variable = { VEN.VEN_caracas_nights_fakemission_days = 80 }
	set_variable = { VEN.VEN_caracas_nights_fakemission_2_days = 7 }
	set_variable = { global.TNO_caracas_coup_control_var^1 = 14 }
	set_variable = { global.TNO_caracas_coup_control_var^2 = 8 }
	set_variable = { global.TNO_caracas_coup_control_var^3 = 18 }
	set_variable = { global.TNO_caracas_coup_control_var^4 = 5 }
	set_variable = { global.TNO_caracas_coup_control_var^5 = 16 }

	TNO_caracas_nights_determine_frame = yes
}

TNO_caracas_nights_change_control = {
	if = {
		limit = {
			check_variable = { VEN_caracas_nights_military_temp_var < 0 }
		}
		custom_effect_tooltip = TNO_caracas_coup_decrease_control_tt
	}
	else_if = {
		limit = {
			check_variable = { VEN_caracas_nights_military_temp_var > 0 }
		}
		custom_effect_tooltip = TNO_caracas_coup_increase_control_tt
	}
	add_to_variable = { global.TNO_caracas_coup_control_var^var:VEN_caracas_nights_effect_district = VEN_caracas_nights_military_temp_var }
	for_loop_effect = {
		start = 0
		end = 5
		value = i

		set_temp_variable = { VEN_caracas_nights_effect_district = i }
		clamp_variable = {
			var = global.TNO_caracas_coup_control_var^i
			min = 0
			max = 100
		}
	}
	TNO_caracas_nights_determine_frame = yes
}

TNO_caracas_nights_determine_frame = {
	for_loop_effect = {
		start = 1
		end = 6
		value = i

		if = {
			limit = {
				check_variable = { global.TNO_caracas_coup_control_var^i < 10 }
			}
			set_variable = { global.TNO_caracas_coup_mapframe_var^i = 2000 }
		}
		else_if = {
			limit = {
				check_variable = { global.TNO_caracas_coup_control_var^i < 25 }
			}
			set_variable = { global.TNO_caracas_coup_mapframe_var^i = 3000 }
		}
		else = {
			set_variable = { global.TNO_caracas_coup_mapframe_var^i = 1000 }
		}
	}
}

TNO_caracas_nights_tick = {
	if = {
		limit = {
			has_global_flag = VEN_CN_ongoing
		}
		activate_mission = IBR_CN_time_mission
		hidden_effect = {
			random_list = {
				20 = {
					set_temp_variable = { VEN_caracas_nights_effect_district = 1 }
				}
				20 = {
					set_temp_variable = { VEN_caracas_nights_effect_district = 2 }
				}
				20 = {
					set_temp_variable = { VEN_caracas_nights_effect_district = 3 }
				}
				20 = {
					set_temp_variable = { VEN_caracas_nights_effect_district = 4 }
				}
				20 = {
					set_temp_variable = { VEN_caracas_nights_effect_district = 5 }
				}
			}
		}
		set_temp_variable = { CN_readiness_temp_var_2 = global.ven_coup_military_readiness }
		if = {
			limit = {
				IBR = { is_ai = yes }
			}
			divide_temp_variable = { CN_readiness_temp_var_2 = 12 }
		}
		else = {
			divide_temp_variable = { CN_readiness_temp_var_2 = 22 }
		}
		set_temp_variable = { CN_popularity_temp_var_2 = global.ven_coup_popularity }
		if = {
			limit = {
				IBR = { is_ai = yes }
			}
			divide_temp_variable = { CN_popularity_temp_var_2 = 15 }
		}
		else = {
			divide_temp_variable = { CN_popularity_temp_var_2 = 25 }
		}
		add_to_temp_variable = { VEN_caracas_nights_military_temp_var = CN_popularity_temp_var_2 }
		add_to_temp_variable = { VEN_caracas_nights_military_temp_var = CN_readiness_temp_var_2 }
		round_temp_variable = VEN_caracas_nights_military_temp_var
		VEN = { set_variable = { VEN_caracas_nights_fakemission_2_days = 7 } }
		TNO_caracas_nights_change_control = yes

		if = {
			limit = {
				IBR = { is_ai = yes }
			}
			set_temp_variable = { ven_coup_popularity_temp_var = 5 }
			TNO_caracas_nights_change_popularity = yes
			set_temp_variable = { ven_coup_readiness_temp_var = 5 }
			TNO_caracas_nights_change_readiness = yes
		}
	}
}
