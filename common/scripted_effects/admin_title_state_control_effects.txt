###State Control Effects
## Handles possible admin title interactions when a state switches owners
# Ex:
# if = {
#		limit = {
#			event_target:admin_title_old_owner = {
#				tag = <old owner>
#			}
#			event_target:admin_title_new_owner = {
#				tag = <new owner>
#			}
#		}
#		<effects here>
#	}

##Admin Titles
TNO_admin_title_state_control_effect_Admin_Title_Province = {
	log = "[GetDateText]: [Root.GetName]: TNO_admin_title_state_control_effect_[?current_title.GetTokenKey]; State: [THIS.GetID], FROM: [?event_target:admin_title_old_owner.GetTag], ROOT: [?event_target:admin_title_new_owner.GetTag]"
	if = {
		limit = {
			event_target:admin_title_old_owner = { tag = ITA }
			event_target:admin_title_new_owner = { tag = TUR }
			has_global_flag = ita_tur_ita_lose
		}
		if = {
			limit = { state = 164 }
			set_temp_variable = { assoc_region_temp = token:Assoc_Region_Aegean }
			TNO_adjust_assoc_region = yes
		}
	}
}
