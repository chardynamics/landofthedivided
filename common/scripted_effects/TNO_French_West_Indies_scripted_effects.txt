TNO_setup_french_west_indies_referendum = {
	hidden_effect = {
		FWI = {
			# Set up initial favor, daily tick
			if = {
				limit = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_MEX } }
				set_variable = { FWI_referendum_favor = 49 } #failproof
			}
			else = {
				set_variable = { FWI_referendum_favor = 51 }
			}
			set_variable = { TNO_FWI_referendum_mendesist_modifier = 0 }
			set_variable = { TNO_FWI_referendum_gaullist_modifier = 0 }
		}
		ZZZ = {
			activate_mission = ZZZ_FWI_Referendum_Deadline
		}
		USA = { country_event = { id = FWI_proxy.7 days = 50 } } #Hidden Event that activates the Veto Decision

		USA = {
			set_temp_variable = { Victory_ID = 68 }
			set_temp_variable = { Victory_Outcome = 0 }
			Cold_War_GUI_AddVictory = yes
		}
	}
}

TNO_conclude_french_west_indies_referendum = {
	USA = {
		TNO_fopo_remove_french_west_indies_referendum_tab = yes
	}
	MEX = {
		TNO_fopo_remove_french_west_indies_referendum_tab = yes
	}
	IBR = {
		TNO_fopo_remove_french_west_indies_referendum_tab = yes
	}
	FWI = {
		if = { #Veto Outcome
			limit = {
				USA = { has_country_flag = FWI_Referendum_USA_VETO }
				OR = {
					check_variable = {
						var = FWI.FWI_referendum_favor
						value = 41
						compare = greater_than_or_equals
					}
					has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_VETO }
				}
				NOT = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_USA } }
				NOT = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_MEX } }
			}
			country_event = { id = FWI_skeleton.2 days = 1 }
			USA = {
				country_event = { id = FWI_proxy.11 }
				clr_country_flag = FWI_Referendum_USA_VETO
			}
			MEX = { country_event = { id = FWI_proxy.11 days = 1 } }
			IBR = { country_event = { id = FWI_proxy.11 days = 1 } }

			USA = {
				set_temp_variable = { Victory_ID = 68 }
				set_temp_variable = { Victory_Outcome = 4 }
				set_temp_variable = { Victory_Points = 50 }
				Cold_War_GUI_ManipulateVictory = yes
			}
		}
		else_if = { #Veto Cancelled Outcome
			limit = {
				USA = { has_country_flag = FWI_Referendum_USA_VETO }
				check_variable = {
					var = FWI.FWI_referendum_favor
					value = 41
					compare = less_than
				}
				NOT = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_USA } }
			}
			country_event = { id = FWI_skeleton.10 days = 1 }
			USA = {
				country_event = { id = FWI_proxy.10 }
				clr_country_flag = FWI_Referendum_USA_VETO
			}
			MEX = { country_event = { id = FWI_proxy.8 } }
			IBR = { country_event = { id = FWI_proxy.9 } }			

			USA = {
				set_temp_variable = { Victory_ID = 68 }
				set_temp_variable = { Victory_Outcome = 2 }
				set_temp_variable = { Victory_Points = -75 }
				Cold_War_GUI_ManipulateVictory = yes
			}
		}
		else_if = { #Confederation Plan
			limit = {
				OR = {
					check_variable = {
						var = FWI.FWI_referendum_favor
						value = 50
						compare = greater_than_or_equals
					}
					has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_USA }
				}
				NOT = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_MEX } }
			}
			country_event = { id = FWI_skeleton.4 days = 1 }
			USA = { country_event = { id = FWI_proxy.5 } }
			MEX = { country_event = { id = FWI_proxy.5 } }
			IBR = { country_event = { id = FWI_proxy.5 } }

			USA = {
				set_temp_variable = { Victory_ID = 68 }
				set_temp_variable = { Victory_Outcome = 5 }
				set_temp_variable = { Victory_Points = 75 }
				Cold_War_GUI_ManipulateVictory = yes
			}
		}
		else = { #Autonomy Plan
			country_event = { id = FWI_skeleton.10 days = 1 }
			USA = { country_event = { id = FWI_proxy.6 } }
			MEX = { country_event = { id = FWI_proxy.6 } }
			IBR = { country_event = { id = FWI_proxy.6 } }

			USA = {
				set_temp_variable = { Victory_ID = 68 }
				set_temp_variable = { Victory_Outcome = 2 }
				set_temp_variable = { Victory_Points = -75 }
				Cold_War_GUI_ManipulateVictory = yes
			}
		}
	}
}

IBR_FWI_referendum_increase_albrand_backing = {
	add_to_variable = { IBR_FWI_referendum_albrand_var = IBR_FWI_referendum_albrand_change }
	clamp_variable = {
		var = IBR_FWI_referendum_albrand_var
		min = 0
		max = 5
	}
	set_variable = { IBR_FWI_referendum_albrand_cost = 75 }
	set_temp_variable = { IBR_FWI_referendum_albrand_tmp = IBR_FWI_referendum_albrand_var }
	multiply_temp_variable = { IBR_FWI_referendum_albrand_tmp = 15 }
	subtract_from_variable = { IBR_FWI_referendum_albrand_cost = IBR_FWI_referendum_albrand_tmp }
	custom_effect_tooltip = IBR_FWI_referendum_increase_albrand_backing_tt
}

TNO_FWI_referendum_calculate_favor = {
	FWI = {
		clamp_variable = {
			var = FWI_referendum_favor
			min = 0
			max = 100
		}
	}
	set_variable = { USA_FWI_confed_cost = FWI.FWI_referendum_favor }
	divide_variable = { USA_FWI_confed_cost = 100 }
	subtract_from_variable = { USA_FWI_confed_cost = 60 }
	multiply_variable = { USA_FWI_confed_cost = -1 }
	if = {
		limit = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_USA } }
		set_variable = { FWI_referendum_favor = 100 }
	}
	else_if = {
		limit = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_VETO } }
		set_variable = { FWI_referendum_favor = 45 }
	}
	else_if = {
		limit = { has_game_rule = { rule = FWI_PROXY option = FWI_PROXY_MEX } }
		set_variable = { FWI_referendum_favor = 0 }
	}
}
