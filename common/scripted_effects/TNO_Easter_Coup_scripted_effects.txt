TNO_fopo_ven_update_coup = {
	clamp_variable = {
		var = global.ven_coup_progress
		min = 0
		max = 100
	}
	clamp_variable = {
		var = global.ven_coup_secrecy
		min = 0
		max = 100
	}
	TNO_fopo_ven_update_coup_phase_increase = yes
	TNO_fopo_ven_update_coup_phase_decrease = yes
}

TNO_fopo_ven_update_coup_phase_increase = {
	if = {
		limit = {
			check_variable = {
				var = global.ven_coup_progress
				value = 10
				compare = greater_than_or_equals
			}
		}
		set_variable = { global.ven_road_frame^0 = 1 }
	}
	if = {
		limit = {
			check_variable = {
				var = global.ven_coup_progress
				value = 20
				compare = greater_than_or_equals
			}
		}
		set_variable = { global.ven_road_frame^1 = 1 }

	if = {

				value = 30

TNO_VEN_easter_coup_launch = {
	clr_global_flag = VEN_easter_coup_ongoing
	if = { # Coup fails
		limit = {
			OR = {
				check_variable = { global.ven_coup_secrecy = 0 }
				check_variable = { global.ven_coup_progress < 85 }
			}
		}
		set_global_flag = VEN_EC_gov_victory
		hidden_effect = {
			BRA = { country_event = BRA_VEN.5 }
			MEX = { country_event = MEX_VEN.13 }
			IBR = { country_event = IBR_VEN.6 }
		}
	}
	if = { # Coup succeeds
		limit = {
			AND = {
				check_variable = { global.ven_coup_secrecy > 0 }
				check_variable = { global.ven_coup_progress > 84.99 }
			}
		}
		set_global_flag = VEN_EC_coup_victory
		hidden_effect = {
			BRA = { country_event = BRA_VEN.7 }
			MEX = { country_event = MEX_VEN.16 }
			IBR = { country_event = IBR_VEN.9 }
		}

		set_temp_variable = { tension_WT = 5 }
		set_temp_variable = { tension_superpower_1 = GER }
		set_temp_variable = { tension_superpower_2 = USA }
		set_temp_variable = { tension_id = 60 }			
		TNO_change_tension = yes
	}
	VEN = {
		if = {
			limit = { has_country_flag = VEN_Barrios_1963 }
			country_event = VEN_skeleton.3
		}
		if = {
			limit = {
				OR = {
					has_country_flag = VEN_Ojeda_1963
					has_country_flag = VEN_Leoni_1963
				}
			}
			country_event = VEN_skeleton.5
		}
	}
	MEX = { TNO_fopo_remove_easter_coup_tab = yes }
	IBR = { TNO_fopo_remove_easter_coup_tab = yes }
}
