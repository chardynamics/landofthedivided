defined_text = {
	name = AdminTitle_GUI_GetName
	text = {
		trigger = {
			has_variable = AdminTitle
			NOT = {
				AND = {
					has_variable = AssocRegion
					is_in_array = { global.is_admin_region = AssocRegion }
				}
			}
		}
		localization_key = "[AdminTitle_GUI_Get_AdminTitle]"
	}
	text = {
		trigger = {
			OR = {
				AND = {
					has_variable = AssocRegion
					is_in_array = { global.is_admin_region = AssocRegion }
				}
				AND = {
					has_variable = AssocRegion
					NOT = { is_in_array = { global.is_admin_region = AssocRegion } }
					NOT = {
						has_variable = AdminTitle

			}

defined_text = {
	name = AdminTitle_MapMode_Get_Same_Diff
	text = {
		trigger = {
			FROM = { state = var:PREV.TNO_admin_title_map_mode_selected_state }
		}
		localization_key = "tno_admin_title_map_mode_same_state_tooltip"
	}
	text = {
		trigger = {
			FROM = { NOT = { has_variable = AdminTitle } }
			NOT = {
				AND = {
					FROM = { has_variable = SubAdminTitle }
					check_variable = { FROM.SubAdminTitle = TNO_admin_title_map_mode_selected_state:AdminTitle }
				}
			}
			OR = {
				check_variable = { global.TNO_admin_title_map_mode_config = 1 }
				check_variable = { global.TNO_admin_title_map_mode_config = 2 }
			}
		}
		localization_key = "tno_admin_title_map_mode_no_admin_title_tooltip"
	}
	text = {
		trigger = {
			FROM = { NOT = { has_variable = AssocRegion } }
			check_variable = { global.TNO_admin_title_map_mode_config = 3 }
		}
		localization_key = "tno_admin_title_map_mode_no_assoc_region_tooltip"
	}
	text = {
		trigger = {
			FROM = { NOT = { has_variable = AssocRegion2 } }
			NOT = {
				AND = {
					FROM = { has_variable = AssocRegion }
					check_variable = { FROM.AssocRegion = TNO_admin_title_map_mode_selected_state:AssocRegion2 }
				}
			}
			check_variable = { global.TNO_admin_title_map_mode_config = 4 }
		}
		localization_key = "tno_admin_title_map_mode_no_assoc_region_2_tooltip"
	}
	text = {
