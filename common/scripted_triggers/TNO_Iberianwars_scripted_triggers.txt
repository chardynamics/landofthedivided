IBR_Wars_spain_invalid_faction_check = {
	OR = {
		NOT = {
			has_country_flag = ibw_supporting_first_faction_spain
		}
		AND = {
			has_country_flag = ibw_supporting_first_faction_spain
			OR = {
				AND = {
					is_in_array = { ROOT.Iberian_Wars_supported_factions = SPR.id }
					NOT = {
						country_exists = SPR
					}
				}
				AND = {
					is_in_array = { ROOT.Iberian_Wars_supported_factions = SPS.id }
					NOT = {
						country_exists = SPS
					}
				}
				AND = {
					is_in_array = { ROOT.Iberian_Wars_supported_factions = FSR.id }
					NOT = {
						country_exists = FSR
					}
				}
				AND = {

					NOT = {
						country_exists = FZL
					}
				}
			}
		}
	}
}
