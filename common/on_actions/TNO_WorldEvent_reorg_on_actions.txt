on_actions = {
	on_startup = {
		effect = {
			every_possible_country = {
				limit = { is_dynamic_country = no }
				if = {
					limit = { HasEuropeInterest = yes }
					set_country_flag = Custom_Europe_Interest
					set_variable = { Europe_Checkbox_var = 2 }
				}
				if = {
					limit = { HasAsiaInterest = yes }
					set_country_flag = Custom_Asia_Interest
					set_variable = { Asia_Checkbox_var = 2 }
				}
				if = {
					limit = { HasAmericasInterest = yes }
					set_country_flag = Custom_America_Interest
					set_variable = { America_Checkbox_var = 2 }
				}
				if = {
					limit = { HasAfricaInterest = yes }
					set_country_flag = Custom_Africa_Interest
					set_variable = { Africa_Checkbox_var = 2 }
				}
				if = {
					limit = { HasOceaniaInterest = yes }
					set_country_flag = Custom_Oceania_Interest
					set_variable = { Pacific_Checkbox_var = 2 }
				}
				if = {
					limit = { HasAntarcticaInterest = yes }
					set_country_flag = Custom_Antarctica_Interest
					set_variable = { Antarctica_Checkbox_var = 2 }
				}
			}
		}
	}
}
