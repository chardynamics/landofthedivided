		effect = {
			log = "([FROM.GetTag]) [FROM.GetName] was annexed by ([ROOT.GetTag]) [ROOT.GetName]."

				remove_from_array = {
					array = TNO_trade_nations

				}
			}

				clear_array = TNO_trade_partners
