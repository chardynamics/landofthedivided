	option = { #Coup Roulette Time
		name = EQG_skeleton.5.a
		ai_chance = { base = 10 }
		if = { #FMA Won
			limit = {
				OR = {
					has_global_flag = WAC_SCENARIO_2_FMA_VICTORY

			}

			}

				limit = { SEN = { check_variable = { ECON_SPHERE_LEADER = USA } } }
				econ_join_sphere_USA = yes
				USA = { country_event = { id = EQG_FOR.4 days = 1 } }
			}

				limit = { SEN = { check_variable = { ECON_SPHERE_LEADER = FFR } } }
				econ_join_sphere_FFR = yes
			}
		}

				limit = { SEN = { check_variable = { ECON_SPHERE_LEADER = USA } } }

				USA = { country_event = { id = EQG_FOR.3 days = 1 } }
			}
			if = {
				limit = { SEN = { check_variable = { ECON_SPHERE_LEADER = SEN } } }
				econ_join_sphere_SEN = yes
			}
		}
		else_if = { #PALF Won
			limit = {
				OR = {
					has_global_flag = WAC_SCENARIO_1_PALF_VICTORY
					has_global_flag = WAC_SCENARIO_2_PALF_VICTORY
					has_global_flag = WAC_SCENARIO_3_PALF_VICTORY
				}
			}
			CAO = { add_to_faction = EQG }

				set_autonomy = {

					end_civil_wars = no
				}
			}
			econ_join_sphere_CAO = yes
			add_ideas = PALF_Satellite_peace

				progressivism = 23

				conservatism_1 = 5
