		country_event = { id = CPV_skeleton.4 days = 5 }

	option = {
		trigger = { #Waiting
			AND = {
				NOT = { has_global_flag = WAC_SCENARIO_1_PALF_VICTORY }

				NOT = { has_global_flag = WAC_SCENARIO_2_FMA_VICTORY }

				NOT = { has_global_flag = WAC_SCENARIO_2_WAA_VICTORY }

				NOT = { has_global_flag = WAC_SCENARIO_3_PALF_VICTORY }

			NOT = { check_variable = { CPV_timer = 8 } }
			NOT = {
				OR = {
					AND = {
						POR = { has_war = no }
						POR = { exists = yes }

						RPP = { has_war = no }
