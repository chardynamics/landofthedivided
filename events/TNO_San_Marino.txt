country_event = { #Election pick event
	id = SNM.4
	immediate = { log = "[GetDateText]: [Root.GetName]: event SNM.4" }
	title = SNM.4.t
	desc = SNM.4.desc
	#picture =

	is_triggered_only = yes

	option = {
		name = SNM.4.a
		ai_chance = { factor = 1 }
		if = {
			limit = {
				has_government = communist
			}
			if = {
				limit = {
					NOT = { ITA = { has_government = socialist } }
				}
				country_event = { id = SNM.9 days = 4 }
			}
			else = {
				random_list = {
					2 = {
						country_event = { id = SNM.6 days = 5 }
					}
					2 = {
						country_event = { id = SNM.7 days = 5 }
					}
					1 = {
						country_event = { id = SNM.8 days = 5 }
					}
				}
			}
		}
		else_if = {
			limit = {
				has_government = progressivism
			}
			random_list = {
				2 = {
					country_event = { id = SNM.5 days = 5 }
				}
				2 = {
					country_event = { id = SNM.7 days = 5 }
				}
				1 = {
					country_event = { id = SNM.8 days = 5 }
				}
			}
		}
		else_if = {
			limit = {
				has_government = liberal_conservatism
			}
			random_list = {
				2 = {
					country_event = { id = SNM.5 days = 5 }
				}
				2 = {
					country_event = { id = SNM.6 days = 5 }
				}
				1 = {
					country_event = { id = SNM.8 days = 5 }
				}
			}
		}
		else_if = {
			limit = {
				OR = {
					has_government = paternalism
					has_government = despotism
				}
			}
			random_list = {
				2 = {
					country_event = { id = SNM.5 days = 5 }
				}
				2 = {
					country_event = { id = SNM.6 days = 5 }
				}
				1 = {
					country_event = { id = SNM.7 days = 5 }
				}
			}
		}
	}
}
