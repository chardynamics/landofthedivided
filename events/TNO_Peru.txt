#Odria Self-Coup
country_event = {
	id = peru_skel.2
	title = peru_skel.2.t
	desc = peru_skel.2.desc
	#picture =

	#fire_only_once = yes
	is_triggered_only = yes

	option = { #Sigma Tip #157: Coup Yourself
		name = peru_skel.2.a
		every_country = {
			limit = {
				is_ai = no
				HasSphereAmericasInterest = yes
			}
			news_event = {
				id = LATAM_news.54
			}
		}
		add_country_leader_role = {
			character = PRU_Manuel_Arturo_Odria
			promote_leader = yes
			country_leader = {
				desc = "POLITICS_MANUEL_ARTURO_ODRIA_DESC"
				expire = "1994.4.20"
				ideology = despotism_personalistic_dictatorship_subtype
				traits = { }
				id = -1
			}
		}
		set_politics = {
			ruling_party = despotism
			elections_allowed = no
		}
		PRU_Manuel_Arturo_Odria = {
			set_portraits = {
				civilian = { large = "gfx/leaders/PRU/Portrait_PRU_Manuel_Odria.png" }
			}
		}
		if = {
			limit = {
				OR = {
					AND = {
						country_exists = CLC
						CLC = { check_variable = { ECON_SPHERE_LEADER = IBR } }
					}
					AND = {
						country_exists = COL
						COL = { check_variable = { ECON_SPHERE_LEADER = IBR } }
					}
					AND = {
						country_exists = CLD
						country_exists = IBR
					}
				}
			}
			econ_join_sphere_IBR = yes
			remove_ideas = OFN_Observer
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						country_exists = CLC
						CLC = { check_variable = { ECON_SPHERE_LEADER = JAP } }
					}
					AND = {
						country_exists = CLC
						CLC = { check_variable = { ECON_SPHERE_LEADER = ITA } }
					}
					AND = {
						country_exists = COL
						COL = { check_variable = { ECON_SPHERE_LEADER = ITA } }
					}
				}
			}
			econ_join_sphere_JAP = yes
			remove_ideas = OFN_Observer
		}
		else = {
			econ_join_sphere_USA = yes #Failsafe
		}
		set_temp_variable = { TNO_economy_subtype_temp = token:Econ_Subtype_Corporate_Oligopoly }
		econ_subtype_change = yes
		PRU = {
			set_variable = { TNO_SA_influence^global.UNALIGNED = 100 }
			set_variable = { TNO_SA_influence^global.BRAZIL = 0 }
			set_variable = { TNO_SA_influence^global.ARGENTINA = 0 }
		}
		set_popularities = {
			communist = 6
			socialist = 5
			progressivism = 11
			liberalism = 16
			conservatism = 22
			despotism = 35
			fascism = 5
		}
	}
}
