country_event = { #Align with France
	id = ALG_skeleton.8
	immediate = { log = "[GetDateText]: [Root.GetName]: event ALG_skeleton.8" }
	title = ALG_skeleton.8.t
	desc = ALG_skeleton.8.d

	is_triggered_only = yes
	#fire_only_once = yes

	option = {
		name = ALG_skeleton.8.a
		ai_chance = { factor = 1 }
		trigger = {
			FRA = { has_war = no }
		}
		#Cool cool cool but which France?
		FRA = {
			set_autonomy = {
				target = SLG
				autonomous_state = autonomy_colony
				end_wars = no
				end_civil_wars = no
			}
		}
		if = {
			limit = { FRA = { check_variable = { ECON_SPHERE_LEADER = USA } } }
			SLG = { econ_join_sphere_USA = yes }
		}
		if = {
			limit = { FRA = { check_variable = { ECON_SPHERE_LEADER = GER } } }
			SLG = { econ_join_sphere_GER = yes }
			add_ideas = Pakt_Mitstreiter
		}
		if = {
			limit = { FRA = { check_variable = { ECON_SPHERE_LEADER = ITA } } }
			SLG = {
				econ_join_sphere_ITA = yes
				if = {
					limit = {
						USA = { is_faction_leader = yes }
						ITA = { is_in_faction_with = USA }
					}
					add_ideas = OFN_Dependent_Member
				}
				else_if = {
					limit = {
						JAP = { is_faction_leader = yes }
						ITA = { is_in_faction_with = JAP }
					}
					add_ideas = Sphere_Economic_Dependent
				}
			}
		}
		#Now Restore your Government
		SLG = {
			set_politics = { ruling_party = fascism }
			set_popularities = {
				despotism = 23
				fascism = 57
				paternalism = 20
			}
			set_cosmetic_tag = ALG_french
		}
		every_owned_state = {
			remove_claim_by = FRA
			remove_core_of = FRA
		}
		if = {
			limit = { tag = SLG }
			every_owned_state = {
				remove_claim_by = ALG
				remove_core_of = ALG
			}
		}
		#Chain
		#country_event = { id = ALG_skeleton.10 days = 450 }
	}
	option = { # We'll be French, but not yet
		name = ALG_skeleton.8.b
		trigger = { FRA = { has_war = yes } }
		country_event = { id = ALG_skeleton.8 days = 14 }
	}
}
