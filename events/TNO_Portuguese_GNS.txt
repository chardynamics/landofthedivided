	id = GNS_wars.102
	immediate = { log = "[GetDateText]: [Root.GetName]: event GNS_wars.102" }
	title = GNS_wars.102.t
	desc = GNS_wars.102.d
	picture = GFX_report_event_iberian_wars_6
	trigger = {
		NOT = {
			has_country_flag = GNS_ceasefire_offered
		}
	}

	is_triggered_only = yes

	fire_only_once = yes

	option = {#Accept
		name = GNS_wars.102.a
		ai_chance = {
			base = 1
		}
		hidden_effect = {
			every_country = {
				limit = { is_ai = no }
				news_event = { id = WORLD.30117 days = 1 }
			}
			if = {
				limit = {
					has_war_with = SPR
				}
				white_peace = SPR
				SPR = {
					TNO_Country_Closing_Screen_Show = yes
					IBRWarsResign = yes
				}
				TNO_Country_Closing_Screen_Show = yes
				IBRWarsResign = yes
			}
			else_if = {
				limit = {
					has_war_with = FSR
				}
				white_peace = FSR
				FSR = {
					TNO_Country_Closing_Screen_Show = yes
					IBRWarsResign = yes
				}
				TNO_Country_Closing_Screen_Show = yes
				IBRWarsResign = yes
			}
			else_if = {
				limit = {
					has_war_with = SPS
				}
				white_peace = SPS
				SPS = {
					TNO_Country_Closing_Screen_Show = yes
					IBRWarsResign = yes
				}
				TNO_Country_Closing_Screen_Show = yes
				IBRWarsResign = yes
			}
		}
	}
	option = {#Deny
		name = GNS_wars.102.b
		ai_chance = {
			base = 0
		}
	}
}
